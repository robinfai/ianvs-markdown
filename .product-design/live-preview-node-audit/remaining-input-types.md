# Remaining HTML input types

<input type="email" value="alpha@example.com"> email

<input type="url" value="https://example.com"> url

<input type="tel" value="123-456"> tel

<input type="search" value="needle"> search

<input type="time" value="13:45"> time

<input type="month" value="2026-09"> month

<input type="week" value="2026-W36"> week

<input type="datetime-local" value="2026-09-03T13:45"> datetime

<input type="color" value="#167b82"> color

<input type="file"> file

<input type="hidden" value="secret"> hidden

<input type="submit" value="Submit now"> submit

<input type="reset" value="Reset now"> reset

<input type="button" value="Push me"> button
