from pathlib import Path
from datetime import datetime, timezone
import gzip
import hashlib
import json
import subprocess

root = Path.cwd()
source = root / 'build/aot-ffi-reduction'
dest = root / 'benchmark/results/2026-10-11-aot-callback'
dest.mkdir(parents=True, exist_ok=False)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def archive(path, relative):
    data = path.read_bytes()
    if path.suffix == '.log':
        relative = str(relative) + '.gz'
        data = gzip.compress(data, mtime=0)
    elif path.suffix in ('.dart', '.py'):
        relative = str(relative) + '.txt'
    target = dest / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data)

for path in sorted(source.rglob('*')):
    if path.is_file() and path.suffix in ('.json', '.log', '.txt', '.dart', '.patch'):
        archive(path, path.relative_to(source))
for label in ('view-trace-3.44.8', 'window-window_scope',
              'window-dialog_controller', 'window-dialog_scope'):
    folder = root / 'build/aot-reduction' / label
    for path in sorted(folder.rglob('*')):
        if path.is_file():
            assert path.suffix in ('.json', '.log', '.txt', '.lock'), path
            archive(path, Path(label) / path.relative_to(folder))
archive(root / 'tool/fixtures/ffi_callback_aot.dart', 'repro-final.dart')

checks = []
def verify_logs(value, folder):
    if isinstance(value, dict):
        if 'log' in value and 'logSha256' in value:
            name = value['log']
            log = folder / name
            if log.exists():
                data = log.read_bytes()
            else:
                log = folder / (name + '.gz')
                data = gzip.decompress(log.read_bytes())
            assert digest(data) == value['logSha256'], log
            checks.append(str(log.relative_to(dest)))
        for item in value.values():
            verify_logs(item, folder)
    elif isinstance(value, list):
        for item in value:
            verify_logs(item, folder)
for path in dest.rglob('*.json'):
    verify_logs(json.loads(path.read_text()), path.parent)

for folder in (dest, dest / 'dead-callbacks'):
    for row in json.loads((folder / 'summary.json').read_text()):
        assert digest((folder / (row['name'] + '.dart.txt')).read_bytes()) == row['sourceSha256']
for stem, filename in [('repro-cross-version', 'repro-with-print.dart.txt'),
                       ('repro-final-cross-version', 'repro-final.dart.txt')]:
    for row in json.loads((dest / (stem + '.json')).read_text()):
        assert row['sourceSha256'] == digest((dest / filename).read_bytes())
        assert row['illegalCidFailure'] and row['result']['exitCode'] == 254
for stem, filename in [('experimental-paired', 'repro-with-print.dart.txt'),
                       ('experimental-final-paired', 'repro-final.dart.txt')]:
    rows = json.loads((dest / (stem + '.json')).read_text())
    for row, code in zip(rows, (-6, 0)):
        src = filename if code == -6 else 'live-callback-control.dart.txt'
        assert row['sourceSha256'] == digest((dest / src).read_bytes())
        assert row['frontend']['exitCode'] == 0
        assert [r['exitCode'] for r in row['snapshots']] == [code, 0]
        assert row['snapshots'][1]['artifactExists']
