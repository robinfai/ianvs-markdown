import 'dart:ffi';
final class Rect extends Struct {
  @Double()
  external double x;
}
final class Request extends Struct {
  @Int64()
  external int id;
  external Pointer<NativeFunction<Pointer<Rect> Function(Pointer<Rect>)>> callback;
}
@Native<Void Function(Request)>(symbol: 'unused_probe')
external void nativeProbe(Request request);
void main() { nativeProbe(Struct.create<Request>()); }
