import 'dart:ffi';
final class Rect extends Struct {
  @Double()
  external double x;
}

  @Int64()
  external int id;
  external Pointer<NativeFunction<Pointer<Rect> Function(Pointer<Rect>)>> callback;
}
@Native<Request Function()>(symbol: 'unused_probe')
external Request nativeProbe();
void main() { print(nativeProbe().id); }
